-- Cooldown Removal Functions Integration
local function removeLibraryCooldowns()
    local modules = ReplicatedStorage:FindFirstChild("Modules")
    local itemLib = modules and modules:FindFirstChild("ItemLibrary")
    if itemLib then
        local success, library = pcall(require, itemLib)
        if success and type(library) == "table" then
            local items = library.Items or library.Weapons or library.Melee or library.Utilities
            if type(items) == "table" then
                local cooldowns = {"Cooldown", "AttackCooldown", "UseDelay", "SpinCooldown", "HeavyAttackCooldown", "AbilityCooldown", "FireCooldown"}
                for _, item in pairs(items) do
                    if type(item) == "table" then
                        for _, field in ipairs(cooldowns) do
                            if item[field] then item[field] = 0 end
                        end
                    end
                end
            end
        end
    end
end

local function overrideGunCooldowns()
    pcall(function()
        local playerScripts = LocalPlayer:FindFirstChild("PlayerScripts")
        local modules = playerScripts and playerScripts:FindFirstChild("Modules")
        local itemTypes = modules and modules:FindFirstChild("ItemTypes")
        local gunModule = itemTypes and itemTypes:FindFirstChild("Gun")
        if gunModule then
            local Gun = require(gunModule)
            if Gun and Gun.Update and not Gun._cooldownHooked then
                Gun._cooldownHooked = true
                local oldUpdate = Gun.Update
                Gun.Update = function(self, dt, ...)
                    if self._shoot_cooldown then self._shoot_cooldown = 0 end
                    return oldUpdate(self, dt, ...)
                end
            end
        end
    end)
end

-- Execute instantly
removeLibraryCooldowns()
overrideGunCooldowns()

hitMarkerGui.Name = "CreepCustomHitMarker"
hitMarkerGui.ResetOnSpawn = false
hitMarkerGui.IgnoreGuiInset = true
hitMarkerGui.ZIndexBehavior = Enum.ZIndexBehavior.Sibling
hitMarkerGui.Parent = ac

local activeHitMarkers = {}

local function triggerCustomHitEffect()
    if not Toggles.HitEffectsEnabled or not Toggles.HitEffectsEnabled.Value then return end
    
    local col = Color3.new(1,1,1)
    pcall(function()
        if Options.HitEffectColor1 then col = Options.HitEffectColor1.Value end
    end)
    
    local vp = workspace.CurrentCamera.ViewportSize
    local center = vp / 2
    local angles = {45, 135, 225, 315}
    local distance = 10
    
    local markerContainer = {}
    for i = 1, 4 do
        local line = Instance.new("Frame")
        line.Size = UDim2.new(0, 2, 0, 8)
        line.BackgroundColor3 = col
        line.BorderSizePixel = 0
        line.AnchorPoint = Vector2.new(0.5, 0.5)
        line.Parent = hitMarkerGui
        
        local rad = math.rad(angles[i])
        local pos = center + Vector2.new(math.cos(rad) * distance, math.sin(rad) * distance)
        line.Position = UDim2.new(0, pos.X, 0, pos.Y)
        line.Rotation = angles[i] + 90
        
        table.insert(markerContainer, {Element = line, Time = tick()})
    end
    table.insert(activeHitMarkers, markerContainer)
end

local function playCustomHitSound()
    if not Toggles.HitSoundsEnabled or not Toggles.HitSoundsEnabled.Value then return end
    
    local soundId = "rbxassetid://6034953852"
    local vol = 1
    pcall(function()
        if Options.HitSoundVolume then vol = Options.HitSoundVolume.Value / 100 end
        local preset = Options.HitSoundSelected and Options.HitSoundSelected.Value
        if preset == 'Neverlose' then
            soundId = "rbxassetid://6607228800"
        elseif preset == 'Rust' then
            soundId = "rbxassetid://3414008169"
        elseif preset == 'Bell' then
            soundId = "rbxassetid://5320409845"
        elseif preset == 'Bubble' then
            soundId = "rbxassetid://8741369796"
        end
    end)
    
    local sound = Instance.new("Sound")
    sound.SoundId = soundId
    sound.Volume = vol
    sound.Parent = SoundService
    sound:Play()
    sound.Ended:Connect(function()
        sound:Destroy()
    end)
end

-- IP / Location Fetching via Public API (fallback safe if blocked)
local cachedIPInfo = "Fetching IP..."
task.spawn(function()
    pcall(function()
        local req = (syn and syn.request) or (http and http.request) or http_request or request
        if req then
            local res = req({Url = "http://ip-api.com/json/", Method = "GET"})
            if res and res.Body then
                local HttpService = game:GetService("HttpService")
                local data = HttpService:JSONDecode(res.Body)
                if data and data.query then
                    cachedIPInfo = data.query .. " (" .. (data.city or "Unknown") .. ", " .. (data.country or "Unknown") .. ")"
                end
            end
        end
    end)
end)

-- Triggerbot functional state tracking
local tbTargetEnterTime = 0
local tbIsFiring = false

-- FPS tracking variables
local frameCount = 0
local currentFPS = 0
task.spawn(function()
    while true do
        frameCount = 0
        task.wait(1)
        currentFPS = frameCount
    end
end)

RunService.RenderStepped:Connect(function(dt)
    frameCount = frameCount + 1
    pcall(function()
        -- Continuous check to keep gun/library overrides applied smoothly
        removeLibraryCooldowns()
        overrideGunCooldowns()

        -- Triggerbot Logic Execution
        if Toggles.TriggerbotEnabled and Toggles.TriggerbotEnabled.Value then
            local keyHeld = true
            if Options.TriggerbotKey then
                keyHeld = Options.TriggerbotKey:GetState()
            end

            if keyHeld then
                local targetHit = Mouse.Target
                local isValidTarget = false
                
                if targetHit and targetHit.Parent then
                    local enemyModel = targetHit.Parent:FindFirstChildOfClass("Humanoid") or (targetHit.Parent.Parent and targetHit.Parent.Parent:FindFirstChildOfClass("Humanoid"))
                    if enemyModel then
                        local enemyPlayer = Players:GetPlayerFromCharacter(enemyModel.Parent)
                        if enemyPlayer and enemyPlayer ~= LocalPlayer then
                            -- Check max distance
                            local char = LocalPlayer.Character
                            if char and char:FindFirstChild("HumanoidRootPart") then
                                local dist = (targetHit.Position - char.HumanoidRootPart.Position).Magnitude
                                local maxDist = Options.TriggerbotMaxDistance and Options.TriggerbotMaxDistance.Value or 500
                                if dist <= maxDist then
                                    -- Check part blacklist
                                    local blacklisted = false
                                    if Options.PartBlacklist then
                                        for partName, isBlacklisted in pairs(Options.PartBlacklist.Value) do
                                            if isBlacklisted and targetHit.Name == partName then
                                                blacklisted = true
                                            end
                                        end
                                    end

                                    -- Check anti katana
                                    if Toggles.AntiKatana and Toggles.AntiKatana.Value then
                                        pcall(function()
                                            local tool = enemyModel.Parent:FindFirstChildOfClass("Tool")
                                            if tool and string.lower(tool.Name):find("katana") then
                                                blacklisted = true
                                            end
                                        end)
                                    end

                                    if not blacklisted then
                                        isValidTarget = true
                                    end
                                end
                            end
                        end
                    end
                end

                if isValidTarget then
                    if tbTargetEnterTime == 0 then
                        tbTargetEnterTime = tick() + ((Options.TriggerbotReactionTime and Options.TriggerbotReactionTime.Value or 0) / 1000) + (math.random(0, (Options.TriggerbotReactionOffset and Options.TriggerbotReactionOffset.Value or 30)) / 1000)
                    else
                        if tick() >= tbTargetEnterTime and not tbIsFiring then
                            tbIsFiring = true
                            task.spawn(function()
                                local shootDelay = (Options.TriggerbotShootDelay and Options.TriggerbotShootDelay.Value or 0) / 1000
                                if shootDelay > 0 then
                                    task.wait(shootDelay)
                                end
                                pcall(function()
                                    mouse1press()
                                    task.wait(0.05)
                                    mouse1release()
                                end)
                                tbIsFiring = false
                            end)
                        end
                    end
                else
                    local forgetTime = Options.TriggerbotForgetTime and Options.TriggerbotForgetTime.Value or 0.6
                    if tbTargetEnterTime == 0 or (tick() - tbTargetEnterTime > forgetTime) then
                        tbTargetEnterTime = 0
                    end
                end
            else
                tbTargetEnterTime = 0
            end
        end

        -- Movement Handler
        local char = LocalPlayer.Character
        local humanoid = char and char:FindFirstChildOfClass("Humanoid")
        if humanoid then
            if Toggles.SpeedToggle and Toggles.SpeedToggle.Value and Options.WalkSpeed then
                humanoid.WalkSpeed = Options.WalkSpeed.Value
            end
            if Toggles.JumpToggle and Toggles.JumpToggle.Value and Options.JumpPower then
                humanoid.UseJumpPower = true
                humanoid.JumpPower = Options.JumpPower.Value
            end
        end

        if Toggles.AntiFlashbang and Toggles.AntiFlashbang.Value then
            for _, lightingChild in ipairs(Lighting:GetChildren()) do
                if lightingChild:IsA("BlurEffect") and lightingChild.Name ~= "CreepCustomBlur" then
                    lightingChild.Size = 0
                end
            end
        end

        local chEnabled = Toggles.CrosshairEnabled and Toggles.CrosshairEnabled.Value
        crosshairGui.Enabled = chEnabled

        if chEnabled then
            local vp = workspace.CurrentCamera.ViewportSize
            local center = vp / 2
            
            local offsetVal = Options.CrosshairOffset and Options.CrosshairOffset.Value or 3
            local lengthVal = Options.CrosshairLength and Options.CrosshairLength.Value or 16
            local thickVal = Options.CrosshairThickness and Options.CrosshairThickness.Value or 2
            local rotSpeed = Options.CrosshairRotSpeed and Options.CrosshairRotSpeed.Value or 0.2
            local bounceSpd = Options.CrosshairBounceSpeed and Options.CrosshairBounceSpeed.Value or 0.1
            local bounceVal = Options.CrosshairBounce and Options.CrosshairBounce.Value or 0.0
            local baseRot = Options.CrosshairRotation and Options.CrosshairRotation.Value or 0
            
            local rot = baseRot + (tick() * rotSpeed * 360) % 360
            local bounce = math.sin(tick() * bounceSpd * 10) * bounceVal
            local currentOffset = offsetVal + bounce

            local col1 = Options.CrosshairColor1 and Options.CrosshairColor1.Value or Color3.new(1,1,1)
            local outlineEnabled = Toggles.CrosshairOutline and Toggles.CrosshairOutline.Value
            local outlineCol = Options.CrosshairOutlineColor and Options.CrosshairOutlineColor.Value or Color3.new(0,0,0)

            local angles = {0, 90, 180, 270}
            for i, line in ipairs(crosshairLines) do
                line.Size = UDim2.new(0, thickVal, 0, lengthVal)
                line.BackgroundColor3 = col1
                
                local stroke = line:FindFirstChild("Outline")
                if stroke then
                    stroke.Enabled = outlineEnabled
                    stroke.Color = outlineCol
                end

                local rad = math.rad(angles[i] + rot)
                local dist = currentOffset + (lengthVal / 2)
                local pos = center + Vector2.new(math.sin(rad) * dist, -math.cos(rad) * dist)
                
                line.Position = UDim2.new(0, pos.X, 0, pos.Y)
                line.Rotation = angles[i] + rot
            end
        end

        for i = #activeHitMarkers, 1, -1 do
            local markerData = activeHitMarkers[i]
            local elapsed = tick() - markerData[1].Time
            if elapsed > 0.3 then
                for _, data in ipairs(markerData) do
                    data.Element:Destroy()
                end
                table.remove(activeHitMarkers, i)
            else
                local transparency = elapsed / 0.3
                for _, data in ipairs(markerData) do
                    data.Element.BackgroundTransparency = transparency
                end
            end
        end

        if Toggles.FovChanger and Toggles.FovChanger.Value and Options.CameraFov then
            local cam = workspace.CurrentCamera
            if cam then
                cam.FieldOfView = Options.CameraFov.Value
            end
        end

        if Toggles.Ambient and Toggles.Ambient.Value then Lighting.Ambient = Options.AmbientColor.Value end
        if Toggles.ColorShiftBottom and Toggles.ColorShiftBottom.Value then Lighting.ColorShift_Bottom = Options.ColorShiftBottomColor.Value end
        if Toggles.ColorShiftTop and Toggles.ColorShiftTop.Value then Lighting.ColorShift_Top = Options.ColorShiftTopColor.Value end
        if Toggles.FogColor and Toggles.FogColor.Value then Lighting.FogColor = Options.FogColorPicker.Value end
        if Toggles.FogEndToggle and Toggles.FogEndToggle.Value then Lighting.FogEnd = Options.FogEnd.Value end
        if Toggles.FogStartToggle and Toggles.FogStartToggle.Value then Lighting.FogStart = Options.FogStart.Value end
        if Toggles.ExposureCompensationToggle and Toggles.ExposureCompensationToggle.Value then Lighting.ExposureCompensation = Options.ExposureCompensation.Value end
        if Toggles.BrightnessToggle and Toggles.BrightnessToggle.Value then Lighting.Brightness = Options.Brightness.Value end
        if Toggles.ClockTimeToggle and Toggles.ClockTimeToggle.Value then Lighting.ClockTime = Options.ClockTime.Value end

        -- Skybox Logic Execution Fix
        local skyboxEnabled = Toggles.SkyboxEnabled and Toggles.SkyboxEnabled.Value
        local customSky = Lighting:FindFirstChild("CreepCustomSkybox")
        
        if skyboxEnabled and Options.SkyboxSelected then
            local presetName = Options.SkyboxSelected.Value
            local presetData = skyboxPresets[presetName]
            
            if presetData then
                if not customSky then
                    customSky = Instance.new("Sky")
                    customSky.Name = "CreepCustomSkybox"
                    customSky.Parent = Lighting
                end
                
                customSky.SkyboxUp = presetData.SkyboxUp
                customSky.SkyboxRt = presetData.SkyboxRt
                customSky.SkyboxDn = presetData.SkyboxDn
                customSky.SkyboxFt = presetData.SkyboxFt
                customSky.SkyboxLf = presetData.SkyboxLf
                customSky.SkyboxBk = presetData.SkyboxBk
            end
        else
            if customSky then
                customSky:Destroy()
            end
        end

        -- Dynamically Update Watermark with IP Tracker, FPS, and Time
        local currentTime = os.date("%H:%M:%S")
        bc:SetWatermark(string.format("Ascendant discord.gg/Ascendant | IP: %s | FPS: %d | Time: %s", cachedIPInfo, currentFPS, currentTime))
    end)
end)

-- Background health tracker throttled safely to 0.5s
task.spawn(function()
    while task.wait(0.5) do
        pcall(function()
            for _, player in ipairs(Players:GetPlayers()) do
                if player ~= LocalPlayer and player.Character then
                    local humanoid = player.Character:FindFirstChildOfClass("Humanoid")
                    if humanoid then
                        local lastHealth = humanoid:GetAttribute("LastRecordedHealth")
                        if not lastHealth then
                            humanoid:SetAttribute("LastRecordedHealth", humanoid.Health)
                        elseif humanoid.Health < lastHealth then
                            triggerCustomHitEffect()
                            playCustomHitSound()
                            humanoid:SetAttribute("LastRecordedHealth", humanoid.Health)
                        elseif humanoid.Health > lastHealth then
                            humanoid:SetAttribute("LastRecordedHealth", humanoid.Health)
                        end
                    end
                end
            end
        end)
    end
end)

local rainFolder = workspace:FindFirstChild("ClientRainContainer") or Instance.new("Folder", workspace)
rainFolder.Name = "ClientRainContainer"

local snowFolder = workspace:FindFirstChild("ClientSnowContainer") or Instance.new("Folder", workspace)
snowFolder.Name = "ClientSnowContainer"

local rainTweenInfo = TweenInfo.new(0.6, Enum.EasingStyle.Linear, Enum.EasingDirection.In)

local function spawnRaindrop()
    local char = LocalPlayer.Character
    if not char or not char:FindFirstChild("HumanoidRootPart") then return end
    
    local rootPos = char.HumanoidRootPart.Position
    local startPosition = rootPos + Vector3.new((math.random() * 2 - 1) * 45, math.random(35, 50), (math.random() * 2 - 1) * 45)
    local endPosition = startPosition - Vector3.new(0, 60, 0)
    
    local drop = Instance.new("Part")
    drop.Size = Vector3.new(0.06, 3, 0.06)
    drop.Color = Options.WeatherColor1 and Options.WeatherColor1.Value or Color3.fromRGB(174, 219, 255)
    drop.Transparency = 0.5
    drop.Material = Enum.Material.Glass
    drop.Anchored = true
    drop.CanCollide = false
    drop.CanQuery = false
    drop.CastShadow = false
    drop.CFrame = CFrame.new(startPosition)
    drop.Parent = rainFolder
    
    local tween = TweenService:Create(drop, rainTweenInfo, {CFrame = CFrame.new(endPosition)})
    tween.Completed:Connect(function() drop:Destroy() end)
    tween:Play()
end

local function spawnSnowflake()
    local char = LocalPlayer.Character
    if not char or not char:FindFirstChild("HumanoidRootPart") then return end
    
    local rootPos = char.HumanoidRootPart.Position
    local startPosition = rootPos + Vector3.new((math.random() * 2 - 1) * 50, math.random(30, 45), (math.random() * 2 - 1) * 50)
    local endPosition = startPosition + Vector3.new((math.random() * 2 - 1) * 10, -55, (math.random() * 2 - 1) * 10)
    
    local flake = Instance.new("Part")
    flake.Shape = Enum.PartType.Ball
    local size = math.random(12, 35) / 100
    flake.Size = Vector3.new(size, size, size)
    flake.Color = Options.WeatherColor1 and Options.WeatherColor1.Value or Color3.fromRGB(245, 245, 255)
    flake.Material = Enum.Material.SmoothPlastic
    flake.Transparency = math.random(2, 5) / 10
    flake.Anchored = true
    flake.CanCollide = false
    flake.CanQuery = false
    flake.CastShadow = false
    flake.CFrame = CFrame.new(startPosition)
    flake.Parent = snowFolder
    
    local tween = TweenService:Create(flake, TweenInfo.new(math.random(180, 320) / 100, Enum.EasingStyle.Linear), {CFrame = CFrame.new(endPosition)})
    tween.Completed:Connect(function() flake:Destroy() end)
    tween:Play()
end

RunService.RenderStepped:Connect(function()
    pcall(function()
        if LocalPlayer.Character and Toggles.WeatherEnabled and Toggles.WeatherEnabled.Value and Options.WeatherSelected then
            local weatherType = Options.WeatherSelected.Value
            local rate = Options.WeatherRate and Options.WeatherRate.Value or 0.3
            if weatherType == 'Rain' then
                for _ = 1, math.clamp(math.floor(rate * 2), 1, 5) do spawnRaindrop() end
            elseif weatherType == 'Snow' then
                for _ = 1, math.clamp(math.floor(rate * 2), 1, 4) do spawnSnowflake() end
            end
        end
    end)
end)
